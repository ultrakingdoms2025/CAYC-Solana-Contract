    ```typescript
    #!/usr/bin/env -S tsx
    /**
     * Resize assets/logo.png (5863x4529 source) to 512x512 + 1024x1024 PNG derivatives
     * via sharp, then run oxipng -o4 for lossless compression.
     *
     * Target: each derivative < 100 KB (wallet-friendly); sharp's PNG output from a
     * clean source typically lands at ~30-80 KB at 512x512 after oxipng -o4.
     *
     * Preserves original as assets/logo-source.png (rename — git sees move+modify).
     * Throws if any output file exceeds 100 KB.
     */
    import sharp from 'sharp';
    import { execFileSync } from 'node:child_process';
    import { existsSync, renameSync, statSync } from 'node:fs';
    import { resolve } from 'node:path';

    const SRC_ORIGINAL = resolve('assets/logo.png');
    const SRC_RENAMED = resolve('assets/logo-source.png');
    const OUT_512 = resolve('assets/logo-512.png');
    const OUT_1024 = resolve('assets/logo-1024.png');
    const MAX_BYTES = 100 * 1024; // 100 KB

    // Step 1: preserve original via rename (idempotent)
    if (existsSync(SRC_ORIGINAL) && !existsSync(SRC_RENAMED)) {
      renameSync(SRC_ORIGINAL, SRC_RENAMED);
      console.log(`Renamed ${SRC_ORIGINAL} -> ${SRC_RENAMED}`);
    }
    if (!existsSync(SRC_RENAMED)) {
      throw new Error(`Source ${SRC_RENAMED} not found. Aborting.`);
    }

    // Step 2: generate 512 + 1024 via sharp
    await sharp(SRC_RENAMED)
      .resize(512, 512, { fit: 'contain', background: { r: 0, g: 0, b: 0, alpha: 0 } })
      .png({ compressionLevel: 9, palette: true })
      .toFile(OUT_512);
    await sharp(SRC_RENAMED)
      .resize(1024, 1024, { fit: 'contain', background: { r: 0, g: 0, b: 0, alpha: 0 } })
      .png({ compressionLevel: 9, palette: true })
      .toFile(OUT_1024);

    // Step 3: oxipng pass for additional lossless gains
    // Use oxipng-bin's exported binary path; falls back to skip if not resolvable.
    let oxipngBin: string | null = null;
    try {
      const mod = await import('oxipng-bin');
      oxipngBin = (mod as any).default ?? (mod as any).oxipngPath ?? null;
    } catch {
      // continue without oxipng — sharp compressionLevel:9 already close to optimal
    }
    if (oxipngBin) {
      execFileSync(oxipngBin, ['-o', '4', '--strip', 'safe', OUT_512, OUT_1024], { stdio: 'inherit' });
    }

    // Step 4: size assertions
    for (const f of [OUT_512, OUT_1024]) {
      const size = statSync(f).size;
      console.log(`${f}: ${size} bytes`);
      if (size > MAX_BYTES) {
        throw new Error(`${f} is ${size} bytes > ${MAX_BYTES} (100 KB) cap. Reduce quality or re-source.`);
      }
    }
    console.log('OK: both derivatives under 100 KB.');
    ```
