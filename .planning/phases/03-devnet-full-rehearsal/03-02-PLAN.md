    ```typescript
    #!/usr/bin/env -S tsx
    import { Command } from 'commander';
    import { Connection, PublicKey, Commitment } from '@solana/web3.js';
    import {
      getMint,
      getExtensionTypes,
      getPermanentDelegate,
      getMetadataPointerState,
      ExtensionType,
      TOKEN_2022_PROGRAM_ID,
    } from '@solana/spl-token';
    import { getTokenMetadata } from '@solana/spl-token-metadata';
    import {
      REHEARSAL_1_METADATA,
      REHEARSAL_2_METADATA,
      TOKEN_DECIMALS,
    } from '../../src/config/token-config.js';
    import { buildConnection, verifyVaultAuthority } from '../../src/squads/index.js';

    export interface VerifyArgs {
      connection: Connection;
      mint: PublicKey;
      expectedVault: PublicKey;
      rehearsal: 1 | 2;
    }

    export async function runVerify(args: VerifyArgs): Promise<{ ok: boolean; errors: string[] }> {
      const errors: string[] = [];
      const expected = args.rehearsal === 1 ? REHEARSAL_1_METADATA : REHEARSAL_2_METADATA;

      const mint = await getMint(args.connection, args.mint, 'confirmed', TOKEN_2022_PROGRAM_ID);

      // 1. decimals
      if (mint.decimals !== TOKEN_DECIMALS) {
        errors.push(`decimals: expected ${TOKEN_DECIMALS}, got ${mint.decimals}`);
      }
      // 2. mint authority
      if (!mint.mintAuthority || !mint.mintAuthority.equals(args.expectedVault)) {
        errors.push(`mintAuthority: expected ${args.expectedVault.toBase58()}, got ${mint.mintAuthority?.toBase58() ?? 'null'}`);
      }
      // 3. freeze authority
      if (!mint.freezeAuthority || !mint.freezeAuthority.equals(args.expectedVault)) {
        errors.push(`freezeAuthority: expected ${args.expectedVault.toBase58()}, got ${mint.freezeAuthority?.toBase58() ?? 'null'}`);
      }
      // 4. extensions
      const exts = getExtensionTypes(mint.tlvData);
      if (!exts.includes(ExtensionType.MetadataPointer)) errors.push('extension missing: MetadataPointer');
      if (!exts.includes(ExtensionType.PermanentDelegate)) errors.push('extension missing: PermanentDelegate');
      // 5. permanent delegate
      const pd = getPermanentDelegate(mint);
      if (!pd?.delegate || !pd.delegate.equals(args.expectedVault)) {
        errors.push(`permanentDelegate: expected ${args.expectedVault.toBase58()}, got ${pd?.delegate?.toBase58() ?? 'null'}`);
      }
      // 6. metadata pointer self-reference
      const mp = getMetadataPointerState(mint);
      if (!mp?.metadataAddress || !mp.metadataAddress.equals(args.mint)) {
        errors.push(`metadataPointer: expected self (${args.mint.toBase58()}), got ${mp?.metadataAddress?.toBase58() ?? 'null'}`);
      }
      // 7. TokenMetadata
      const md = await getTokenMetadata(args.connection, args.mint, 'confirmed', TOKEN_2022_PROGRAM_ID);
      if (!md) {
        errors.push('TokenMetadata: expected initialized, got null');
      } else {
        if (md.name !== expected.name) errors.push(`name: expected ${expected.name}, got ${md.name}`);
        if (md.symbol !== expected.symbol) errors.push(`symbol: expected ${expected.symbol}, got ${md.symbol}`);
        if (!md.updateAuthority || !md.updateAuthority.equals(args.expectedVault)) {
          errors.push(`updateAuthority: expected ${args.expectedVault.toBase58()}, got ${md.updateAuthority?.toBase58() ?? 'null'}`);
        }
      }

      return { ok: errors.length === 0, errors };
    }

    // ---- CLI wrapper ----
    // Guard so import-for-tests does not trigger CLI parsing
    const isCli = import.meta.url === `file://${process.argv[1]}` || process.argv[1]?.endsWith('verify-mint.ts');
    if (isCli) {
      const program = new Command();
      program
        .requiredOption('--network <net>', 'devnet|mainnet-beta')
        .requiredOption('--mint <pubkey>', 'mint address to verify')
        .requiredOption('--expected-vault <pubkey>', 'expected Squads vault PDA (authority target)')
        .requiredOption('--rehearsal <n>', 'rehearsal bundle (1 or 2)', (v) => {
          const n = Number(v);
          if (n !== 1 && n !== 2) throw new Error('--rehearsal must be 1 or 2');
          return n as 1 | 2;
        })
        .parse(process.argv);

      const opts = program.opts();
      const network = opts.network as 'devnet' | 'mainnet-beta';
      const connection = buildConnection(network, 'confirmed');
      const mint = new PublicKey(opts.mint);
      const expectedVault = new PublicKey(opts.expectedVault);
      const rehearsal = opts.rehearsal as 1 | 2;

      const result = await runVerify({ connection, mint, expectedVault, rehearsal });
      if (result.ok) {
        console.log(`PASS: ${mint.toBase58()} matches all expected fields for rehearsal ${rehearsal}`);
        process.exit(0);
      } else {
        console.error(`FAIL: ${mint.toBase58()} has ${result.errors.length} mismatches:`);
        for (const e of result.errors) console.error(`  - ${e}`);
        process.exit(1);
      }
    }
    ```
